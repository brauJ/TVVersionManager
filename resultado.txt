generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "mysql"
  url      = env("DATABASE_URL")
}

enum UserStatus {
  ACTIVE
  DISABLED
}

enum EmployeeStatus {
  ACTIVE
  INACTIVE
}

enum AttendanceType {
  ENTRY
  BREAK_START
  BREAK_END
  EXIT
}

enum AttendanceMethod {
  FINGERPRINT
  PIN
  ADMIN_MANUAL
}

enum AttendanceStatus {
  CONFIRMED
  PENDING_REVIEW
  VOIDED
}

enum ShiftMode {
  TWO_PUNCH
  FOUR_PUNCH
}

model User {
  id           String     @id @default(cuid())
  email        String     @unique @db.VarChar(190)
  passwordHash String     @db.VarChar(255)
  fullName     String     @db.VarChar(190)
  status       UserStatus @default(ACTIVE)
  isSuperAdmin Boolean    @default(false)
  permissions  UserPermission[]
  auditLogs    AuditLog[]
  createdAt    DateTime   @default(now())
  updatedAt    DateTime   @updatedAt
}

model Permission {
  id          String           @id @default(cuid())
  code        String           @unique @db.VarChar(100)
  description String           @db.VarChar(255)
  users       UserPermission[]
}

model UserPermission {
  userId       String
  permissionId String
  user         User       @relation(fields: [userId], references: [id], onDelete: Cascade)
  permission   Permission @relation(fields: [permissionId], references: [id], onDelete: Cascade)

  @@id([userId, permissionId])
}

model Department {
  id        String     @id @default(cuid())
  name      String     @unique @db.VarChar(120)
  active    Boolean    @default(true)
  employees Employee[]
}

model Branch {
  id        String     @id @default(cuid())
  name      String     @unique @db.VarChar(120)
  address   String?    @db.VarChar(255)
  active    Boolean    @default(true)
  employees Employee[]
}

model Employee {
  id                    String                 @id @default(cuid())
  rut                   String                 @unique @db.VarChar(20)
  firstName             String                 @db.VarChar(100)
  lastName              String                 @db.VarChar(100)
  position              String?                @db.VarChar(120)
  departmentId          String?
  branchId              String?
  hiredAt               DateTime?
  status                EmployeeStatus          @default(ACTIVE)
  department             Department?             @relation(fields: [departmentId], references: [id], onDelete: SetNull)
  branch                 Branch?                 @relation(fields: [branchId], references: [id], onDelete: SetNull)
  shiftAssignments       ShiftAssignment[]
  sessions               WorkSession[]
  attendance             AttendanceRecord[]
  biometricCredentials   BiometricCredential[]
  createdAt              DateTime               @default(now())
  updatedAt              DateTime               @updatedAt

  @@index([lastName, firstName])
  @@index([status])
}

model Shift {
  id               String            @id @default(cuid())
  name             String            @unique @db.VarChar(120)
  startTime        String            @db.VarChar(5)
  endTime          String            @db.VarChar(5)
  breakStart       String?           @db.VarChar(5)
  breakEnd         String?           @db.VarChar(5)
  toleranceMinutes Int               @default(0)
  mode             ShiftMode         @default(FOUR_PUNCH)
  crossesMidnight  Boolean           @default(false)
  active           Boolean           @default(true)
  assignments      ShiftAssignment[]
  createdAt        DateTime          @default(now())
}

model ShiftAssignment {
  id         String   @id @default(cuid())
  employeeId String
  shiftId    String
  validFrom  DateTime
  validTo    DateTime?
  employee   Employee @relation(fields: [employeeId], references: [id])
  shift      Shift    @relation(fields: [shiftId], references: [id])

  @@index([employeeId, validFrom, validTo])
}

model WorkSession {
  id         String             @id @default(cuid())
  employeeId String
  workDate   DateTime           @db.Date
  shiftId    String?
  employee   Employee           @relation(fields: [employeeId], references: [id])
  records    AttendanceRecord[]
  createdAt  DateTime           @default(now())

  @@unique([employeeId, workDate])
}

model AttendanceRecord {
  id              String                  @id @default(cuid())
  eventId         String                  @unique @db.VarChar(190)
  employeeId      String
  sessionId       String?
  type            AttendanceType
  method          AttendanceMethod
  status          AttendanceStatus         @default(CONFIRMED)
